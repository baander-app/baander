cmake_minimum_required(VERSION 3.24)
project(baander_registry VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
option(REGISTRY_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(REGISTRY_THREAD_SANITIZER "Enable separate ThreadSanitizer qualification" OFF)
option(REGISTRY_FUZZ "Enable Clang libFuzzer parser qualification with ASan/UBSan" OFF)
if(REGISTRY_FUZZ)
  if(REGISTRY_THREAD_SANITIZER)
    message(FATAL_ERROR "Parser fuzzing must run separately from ThreadSanitizer")
  endif()
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Parser fuzzing requires Clang with libFuzzer support")
  endif()
  set(REGISTRY_SANITIZERS ON)
endif()
if(REGISTRY_SANITIZERS AND REGISTRY_THREAD_SANITIZER)
  message(FATAL_ERROR "ThreadSanitizer must run separately from address/undefined sanitizers")
endif()
find_package(OpenSSL 3.5.3 EXACT REQUIRED)
find_package(Threads REQUIRED)
FetchContent_Declare(boost_headers URL https://archives.boost.io/release/1.88.0/source/boost_1_88_0.tar.bz2 SOURCE_SUBDIR registry_headers_only URL_HASH SHA256=46d9d2c06637b219270877c9e16155cbd015b6dc84349af064c088e9b5b12f7b DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
# The full pinned Boost archive supplies header-only Asio/Beast and Boost.URL's
# compiled Boost.URL sources; no system Boost substitution.
FetchContent_MakeAvailable(boost_headers)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
file(GLOB_RECURSE BOOST_URL_SOURCES CONFIGURE_DEPENDS ${boost_headers_SOURCE_DIR}/libs/url/src/*.cpp)
add_library(registry_boost_url STATIC ${BOOST_URL_SOURCES})
target_include_directories(registry_boost_url SYSTEM PUBLIC ${boost_headers_SOURCE_DIR})
target_compile_definitions(registry_boost_url PUBLIC BOOST_URL_STATIC_LINK=1 BOOST_URL_NO_LIB=1 PRIVATE BOOST_URL_SOURCE)
add_library(registry_core cpp/core.cpp)
target_include_directories(registry_core PUBLIC include)
target_include_directories(registry_core SYSTEM PUBLIC ${boost_headers_SOURCE_DIR})
target_link_libraries(registry_core PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto OpenSSL::SSL Threads::Threads registry_boost_url)
target_compile_options(registry_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(REGISTRY_SANITIZERS)
  target_compile_options(registry_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(registry_core PUBLIC -fsanitize=address,undefined)
endif()
if(REGISTRY_THREAD_SANITIZER)
  target_compile_options(registry_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer -g)
  target_link_options(registry_core PUBLIC -fsanitize=thread)
endif()
include(CTest)
if(BUILD_TESTING)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.17.0 URL_HASH SHA256=65fab701d9829d38cb77c14acdc431d2108bfdbf8979e40eb8ae567edf10b27c DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googletest)
  add_executable(registry_tests tests/core_test.cpp tests/security_test.cpp)
  target_link_libraries(registry_tests PRIVATE registry_core GTest::gtest_main)
  target_compile_options(registry_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  include(GoogleTest)
  gtest_discover_tests(registry_tests)
  if(REGISTRY_THREAD_SANITIZER)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(registry_tsan_probe tests/tsan_probe.cpp)
    target_link_libraries(registry_tsan_probe PRIVATE Threads::Threads)
    target_compile_options(registry_tsan_probe PRIVATE -Wall -Wextra -Wpedantic -Werror -fsanitize=thread -fno-omit-frame-pointer -g)
    target_link_options(registry_tsan_probe PRIVATE -fsanitize=thread)
    add_test(NAME registry_tsan_positive_control
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_tsan.py
              --probe $<TARGET_FILE:registry_tsan_probe>)
    set_tests_properties(registry_tsan_positive_control PROPERTIES TIMEOUT 30)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(registry_contract_fixture tests/contract_fixture.cpp)
  target_link_libraries(registry_contract_fixture PRIVATE registry_core)
  target_compile_options(registry_contract_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(registry_transport cpp/transport.cpp)
target_link_libraries(registry_transport PUBLIC registry_core)
target_compile_options(registry_transport PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(REGISTRY_FUZZ)
  # Libraries carry coverage instrumentation without introducing a fuzzer main.
  foreach(target registry_core registry_transport registry_boost_url)
    target_compile_options(${target} PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  endforeach()
  add_executable(registry_parser_fuzz tests/parser_fuzz.cpp)
  target_link_libraries(registry_parser_fuzz PRIVATE registry_transport)
  target_compile_options(registry_parser_fuzz PRIVATE -Wall -Wextra -Wpedantic -Werror -fsanitize=fuzzer-no-link)
  target_link_options(registry_parser_fuzz PRIVATE -fsanitize=fuzzer)
endif()
if(BUILD_TESTING)
  add_executable(registry_transport_tests tests/transport_test.cpp)
  target_link_libraries(registry_transport_tests PRIVATE registry_transport GTest::gtest_main)
  target_compile_options(registry_transport_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  gtest_discover_tests(registry_transport_tests)
endif()
if(BUILD_TESTING)
  add_executable(registry_transport_fixture tests/transport_fixture.cpp)
  target_link_libraries(registry_transport_fixture PRIVATE registry_transport)
  target_compile_options(registry_transport_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(baander-registry cpp/server.cpp)
target_link_libraries(baander-registry PRIVATE registry_transport)
target_compile_options(baander-registry PRIVATE -Wall -Wextra -Wpedantic -Werror)
